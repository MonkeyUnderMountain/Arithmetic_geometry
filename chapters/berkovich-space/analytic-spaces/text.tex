\section{Analytic spaces}

    Fix a complete non-archimedean field \(\kk\).

\subsection{Definition of analytic spaces}


    \begin{definition}\label{def:net_and_quasi_net_to_define_analytic_space}
        Let \(X\) be a topological space. 
        A \emph{net} on \(X\) is a collection \(\calN\) of subsets of \(X\) such that
        \begin{enumerate}
            \item for every \(x \in X\), there exist \(U_1, \ldots, U_n \in \calN\) such that \(x \in \bigcap_i U_i\) and \(\bigcup_i U_i\) is an open neighborhood of \(x\) in \(X\);
            \item for every \(U, V \in \calN\), the collection \(\calN|_{U \cap V} \coloneqq \{W \in \calN : W \subseteq U \cap V\} \subset \calP(U \cap V)\) satisfies the first condition for the topological space \(U \cap V\).
        \end{enumerate}
    \end{definition}

    \Yang{For example, if \(X\) is a \(\kk\)-affinoid space, then the collection of all affinoid domains in \(X\) forms a net on \(X\).}

    \begin{definition}\label{def:berkovich_analytic_space_by_net}
        Let \(X\) be a locally Hausdorff topological space.
        A \emph{\(\kk\)-affinoid atlas} on \(X\) is a net \(\calN\) on \(X\) together with
        \begin{enumerate}
            \item for every \(U \in \calN\), a structure of a \(\kk\)-affinoid space on \(U\), i.e. a \(\kk\)-affinoid algebra \(A_U\) and an isomorphism \(U \cong \calM(A_U)\);
            \item for every \(U, V \in \calN\) with \(V \subseteq U\), a morphism \(\res^U_V \colon A_U \to A_V\) of \(\kk\)-affinoid algebras such that the induced morphism \(\calM(A_V) \to \calM(A_U)\) making \(V\) a \(\kk\)-affinoid subdomain in \(U\).
        \end{enumerate}
        A \emph{\(\kk\)-analytic space} or \emph{Berkovich space} is a locally Hausdorff topological space \(X\) together with a \(\kk\)-affinoid atlas on \(X\).
    \end{definition}

    \begin{definition}\label{def:category_of_analytic_space}
        Let \((X, \scrO_X)\) and \((Y, \scrO_Y)\) be \(\kk\)-analytic spaces. 
        A \emph{morphism} \(f \colon (X, \scrO_X) \to (Y, \scrO_Y)\) of \(\kk\)-analytic spaces is a morphism of locally ringed spaces. 
        The category of \(\kk\)-analytic spaces is the category whose objects are \(\kk\)-analytic spaces and whose morphisms are morphisms of \(\kk\)-analytic spaces.
        \Yang{To be revised.}
    \end{definition}

    \begin{theorem}\label{thm:fibre_product_of_analytic_space_existss}
        Let \((X, \scrO_X)\), \((Y, \scrO_Y)\), and \((Z, \scrO_Z)\) be \(\kk\)-analytic spaces, and let \(f \colon (X, \scrO_X) \to (Z, \scrO_Z)\) and \(g \colon (Y, \scrO_Y) \to (Z, \scrO_Z)\) be morphisms of \(\kk\)-analytic spaces. 
        Then the fibre product \((X, \scrO_X) \times_{(Z, \scrO_Z)} (Y, \scrO_Y)\) exists in the category of \(\kk\)-analytic spaces.
        \Yang{To be revised.}
    \end{theorem}

    \begin{definition}\label{def:good_analytic_space}
        Let \((X, \scrO_X)\) be a \(\kk\)-analytic space. 
        We say that \((X, \scrO_X)\) is \emph{good} if every point \(x \in X\) has an open neighborhood \(U\) such that \((U, \scrO_X|_U)\) is a \(\kk\)-affinoid space.
    \end{definition}


\subsection{The first properties}

    \begin{definition}\label{def:finite_morphism_between_analytic_space}
        Let \(f \colon (X, \scrO_X) \to (Y, \scrO_Y)\) be a morphism of \(\kk\)-analytic spaces. 
        We say that \(f\) is \emph{finite} if for every affinoid domain \(V \subseteq Y\), the preimage \(f^{-1}(V)\) is an affinoid domain in \(X\) and the induced morphism \(\scrO_Y(V) \to \scrO_X(f^{-1}(V))\) is a finite morphism of \(\kk\)-affinoid algebras.
        \Yang{}
    \end{definition}

    \begin{theorem}\label{thm:Stein_factoriation_of_analytic_space}
        Let \(f \colon (X, \scrO_X) \to (Y, \scrO_Y)\) be a morphism of \(\kk\)-analytic spaces. 
        Then there exists a factorization \(f = g \circ h\) where \(h \colon (X, \scrO_X) \to (Z, \scrO_Z)\) is a finite morphism and \(g \colon (Z, \scrO_Z) \to (Y, \scrO_Y)\) is a morphism such that for every affinoid domain \(W \subseteq Y\), the preimage \(g^{-1}(W)\) is an affinoid domain in \(Z\) and the induced morphism \(\scrO_Y(W) \to \scrO_Z(g^{-1}(W))\) is a flat morphism of \(\kk\)-affinoid algebras.
        \Yang{}
    \end{theorem}

    \begin{definition}\label{def:seperated_morphism_between_analytic_space}
        Let \(f \colon (X, \scrO_X) \to (Y, \scrO_Y)\) be a morphism of \(\kk\)-analytic spaces. 
        We say that \(f\) is \emph{separated} if the diagonal morphism \(\Delta_f \colon (X, \scrO_X) \to (X, \scrO_X) \times_{(Y, \scrO_Y)} (X, \scrO_X)\) is a closed immersion.
        \Yang{}
    \end{definition}

    \begin{definition}\label{def:proper_morphism_between_analytic_space}
        Let \(f \colon (X, \scrO_X) \to (Y, \scrO_Y)\) be a morphism of \(\kk\)-analytic spaces. 
        We say that \(f\) is \emph{proper} if it is separated, compact, and without boundary.
        \Yang{}
    \end{definition}

    \begin{theorem}\label{thm:direct_image_under_proper_morphism_is_coherent}
        Let \(f \colon (X, \scrO_X) \to (Y, \scrO_Y)\) be a proper morphism of \(\kk\)-analytic spaces. 
        Then for every coherent \(\scrO_X\)-module \(\scrF\), the direct image \(f_*(\scrF)\) is a coherent \(\scrO_Y\)-module.
        \Yang{}
    \end{theorem}
