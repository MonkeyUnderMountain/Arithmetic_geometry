    \begin{proposition}\label{prop:krull_dimension_of_affinoid_domain_under_ground_field_changing}
        Let \(A\) be a strictly \(\kk\)-affinoid algebra.
        For any completed valued field extension \(\KK/\kk\), we have 
        \[ \dim_{\text{Krull}}(A) = \dim_{\text{Krull}}(A \hat{\otimes}_\kk \KK). \]

    \end{proposition}
