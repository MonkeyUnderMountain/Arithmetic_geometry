\section{Analytic spaces}

    Fix a complete non-archimedean field \(\kk\).

\subsection{Definition of analytic spaces}


    \begin{definition}\label{def:k-quasiaffinoid_space}
        A \emph{\(\kk\)-quasiaffinoid space} is a locally ringed space \((X, \scrO_X)\) which is isomorphic to an open subspace of a \(\kk\)-affinoid space.
    \end{definition}

    \begin{definition}\label{def:k-analytic_space}
        A \emph{\(\kk\)-analytic space} is a locally ringed space \((X, \scrO_X)\) such that every point \(x \in X\) has an open neighborhood \(U\) such that \((U, \scrO_X|_U)\) is a \(\kk\)-quasiaffinoid space.
    \end{definition}
